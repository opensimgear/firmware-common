cmake_minimum_required(VERSION 3.24)
project(opensimgear_common LANGUAGES C)
option(OSG_EMBEDDED "Build STM32/TinyUSB dependencies" OFF)
option(OSG_TEST_COVERAGE "Instrument native tests and enforce >70% core coverage" OFF)
if(OSG_TEST_COVERAGE)
  if(OSG_EMBEDDED OR NOT PROJECT_IS_TOP_LEVEL OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Coverage requires a top-level native GNU/Clang build")
  endif()
  add_compile_options(--coverage)
  add_link_options(--coverage)
endif()
add_library(osg_core STATIC src/inputs/button.c src/inputs/inputs.c src/inputs/profile.c src/inputs/tca9555.c src/config/line.c src/actions/actions.c src/events/events.c)
target_include_directories(osg_core PUBLIC include)
target_compile_features(osg_core PUBLIC c_std_11)
target_compile_options(osg_core PRIVATE -Wall -Wextra -Werror)
if(OSG_EMBEDDED)
  if(NOT OSG_USB_CONFIG_DIR)
    message(FATAL_ERROR "OSG_USB_CONFIG_DIR must contain tusb_config.h")
  endif()
  include(cmake/dependencies.cmake)
  add_library(osg_tinyusb STATIC
    ${tinyusb_SOURCE_DIR}/src/tusb.c ${tinyusb_SOURCE_DIR}/src/common/tusb_fifo.c
    ${tinyusb_SOURCE_DIR}/src/device/usbd.c ${tinyusb_SOURCE_DIR}/src/class/cdc/cdc_device.c
    ${tinyusb_SOURCE_DIR}/src/class/hid/hid_device.c
    ${tinyusb_SOURCE_DIR}/src/portable/st/stm32_fsdev/dcd_stm32_fsdev.c
    ${tinyusb_SOURCE_DIR}/src/portable/st/stm32_fsdev/fsdev_common.c)
  target_include_directories(osg_tinyusb PUBLIC ${tinyusb_SOURCE_DIR}/src ${OSG_USB_CONFIG_DIR}
    ${stm32_cmsis_SOURCE_DIR}/Include ${cmsis_core_SOURCE_DIR}/CMSIS/Core/Include)
  target_compile_definitions(osg_tinyusb PUBLIC STM32G0B1xx CFG_TUSB_MCU=OPT_MCU_STM32G0)
  add_library(osg_usb STATIC src/usb/descriptor.c)
  target_link_libraries(osg_usb PUBLIC osg_tinyusb)
  target_include_directories(osg_usb PUBLIC include)
  add_library(osg_canard STATIC ${libcanard_SOURCE_DIR}/libcanard/canard.c)
  target_include_directories(osg_canard PUBLIC ${libcanard_SOURCE_DIR}/libcanard ${libcanard_SOURCE_DIR}/lib/cavl2)
  set(OSG_STM32_HAL_DIR ${stm32_hal_SOURCE_DIR} PARENT_SCOPE)
  set(OSG_STM32_CMSIS_DIR ${stm32_cmsis_SOURCE_DIR} PARENT_SCOPE)
  set(OSG_CMSIS_CORE_DIR ${cmsis_core_SOURCE_DIR} PARENT_SCOPE)
endif()
if(PROJECT_IS_TOP_LEVEL AND NOT OSG_EMBEDDED)
  enable_testing()
  add_executable(osg_events_test tests/events_test.c)

  target_compile_options(osg_events_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)

  add_executable(osg_actions_test tests/actions_test.c)
  target_link_libraries(osg_actions_test PRIVATE osg_core)
  target_compile_options(osg_actions_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME device_action_behavior COMMAND osg_actions_test)
  add_executable(osg_core_test tests/core_test.c)
  target_link_libraries(osg_core_test PRIVATE osg_core)
  target_compile_options(osg_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME core_behavior COMMAND osg_core_test)
  add_executable(osg_inputs_test tests/inputs_test.c)
  target_link_libraries(osg_inputs_test PRIVATE osg_core)
  target_compile_options(osg_inputs_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME controller_input_behavior COMMAND osg_inputs_test)
  add_executable(osg_profile_test tests/profile_test.c)
  target_link_libraries(osg_profile_test PRIVATE osg_core)
  target_compile_options(osg_profile_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME generic_profile_behavior COMMAND osg_profile_test)
endif()

option(OSG_TEST_USB_DESCRIPTORS "Fetch TinyUSB headers for native USB descriptor tests" OFF)
if(PROJECT_IS_TOP_LEVEL AND NOT OSG_EMBEDDED AND OSG_TEST_USB_DESCRIPTORS)
  include(cmake/dependencies.cmake)
  add_executable(osg_usb_test tests/usb_descriptor_test.c src/usb/descriptor.c)
  target_include_directories(osg_usb_test PRIVATE include tests/usb ${tinyusb_SOURCE_DIR}/src)
  target_compile_features(osg_usb_test PRIVATE c_std_11)
  target_compile_options(osg_usb_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME usb_descriptor_behavior COMMAND osg_usb_test)
  add_executable(osg_usb_topology_test tests/usb_topology_test.c src/usb/descriptor.c
    ${tinyusb_SOURCE_DIR}/src/tusb.c ${tinyusb_SOURCE_DIR}/src/common/tusb_fifo.c
    ${tinyusb_SOURCE_DIR}/src/device/usbd.c ${tinyusb_SOURCE_DIR}/src/class/cdc/cdc_device.c
    ${tinyusb_SOURCE_DIR}/src/class/hid/hid_device.c)
  target_include_directories(osg_usb_topology_test PRIVATE include tests/usb ${tinyusb_SOURCE_DIR}/src)
  target_compile_features(osg_usb_topology_test PRIVATE c_std_11)
  target_compile_options(osg_usb_topology_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME usb_topology_behavior COMMAND osg_usb_topology_test)
endif()

if(OSG_TEST_COVERAGE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(APPLE AND CMAKE_C_COMPILER_ID MATCHES "Clang")
    set(OSG_GCOV_COMMAND "xcrun llvm-cov gcov" CACHE STRING "gcov command matching the native compiler")
  else()
    set(OSG_GCOV_COMMAND "gcov" CACHE STRING "gcov command matching the native compiler")
  endif()
  add_custom_target(coverage
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_coverage.py
      --build ${CMAKE_BINARY_DIR} --gcov-command "${OSG_GCOV_COMMAND}"
    VERBATIM)
endif()
