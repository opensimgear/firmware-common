cmake_minimum_required(VERSION 3.24)
project(opensimgear_common LANGUAGES C)
option(OSG_EMBEDDED "Build STM32/TinyUSB dependencies" OFF)
add_library(osg_core STATIC src/inputs/button.c src/config/line.c)
target_include_directories(osg_core PUBLIC include)
target_compile_features(osg_core PUBLIC c_std_11)
target_compile_options(osg_core PRIVATE -Wall -Wextra -Werror)
if(OSG_EMBEDDED)
  if(NOT OSG_USB_CONFIG_DIR)
    message(FATAL_ERROR "OSG_USB_CONFIG_DIR must contain tusb_config.h")
  endif()
  include(cmake/dependencies.cmake)
  add_library(osg_tinyusb STATIC
    ${tinyusb_SOURCE_DIR}/src/tusb.c ${tinyusb_SOURCE_DIR}/src/common/tusb_fifo.c
    ${tinyusb_SOURCE_DIR}/src/device/usbd.c ${tinyusb_SOURCE_DIR}/src/class/cdc/cdc_device.c
    ${tinyusb_SOURCE_DIR}/src/class/hid/hid_device.c
    ${tinyusb_SOURCE_DIR}/src/portable/st/stm32_fsdev/dcd_stm32_fsdev.c
    ${tinyusb_SOURCE_DIR}/src/portable/st/stm32_fsdev/fsdev_common.c)
  target_include_directories(osg_tinyusb PUBLIC ${tinyusb_SOURCE_DIR}/src ${OSG_USB_CONFIG_DIR}
    ${stm32_cmsis_SOURCE_DIR}/Include ${cmsis_core_SOURCE_DIR}/CMSIS/Core/Include)
  target_compile_definitions(osg_tinyusb PUBLIC STM32G0B1xx CFG_TUSB_MCU=OPT_MCU_STM32G0)
  add_library(osg_usb STATIC src/usb/descriptor.c)
  target_link_libraries(osg_usb PUBLIC osg_tinyusb)
  target_include_directories(osg_usb PUBLIC include)
  add_library(osg_canard STATIC ${libcanard_SOURCE_DIR}/libcanard/canard.c)
  target_include_directories(osg_canard PUBLIC ${libcanard_SOURCE_DIR}/libcanard ${libcanard_SOURCE_DIR}/lib/cavl2)
  set(OSG_STM32_HAL_DIR ${stm32_hal_SOURCE_DIR} PARENT_SCOPE)
  set(OSG_STM32_CMSIS_DIR ${stm32_cmsis_SOURCE_DIR} PARENT_SCOPE)
  set(OSG_CMSIS_CORE_DIR ${cmsis_core_SOURCE_DIR} PARENT_SCOPE)
endif()
if(PROJECT_IS_TOP_LEVEL AND NOT OSG_EMBEDDED)
  enable_testing()
  add_executable(osg_core_test tests/core_test.c)
  target_link_libraries(osg_core_test PRIVATE osg_core)
  target_compile_options(osg_core_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME core_behavior COMMAND osg_core_test)
endif()

option(OSG_TEST_USB_DESCRIPTORS "Fetch TinyUSB headers for native USB descriptor tests" OFF)
if(PROJECT_IS_TOP_LEVEL AND NOT OSG_EMBEDDED AND OSG_TEST_USB_DESCRIPTORS)
  include(cmake/dependencies.cmake)
  add_executable(osg_usb_test tests/usb_descriptor_test.c src/usb/descriptor.c)
  target_include_directories(osg_usb_test PRIVATE include tests/usb ${tinyusb_SOURCE_DIR}/src)
  target_compile_features(osg_usb_test PRIVATE c_std_11)
  target_compile_options(osg_usb_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME usb_descriptor_behavior COMMAND osg_usb_test)
endif()
